// TRANSIENT test-phase harness; deleted after the run.
import XCTest
import SwiftUI
@testable import DeskFloater

@MainActor
final class ZZTransientHarness: XCTestCase {
    let ev = ProcessInfo.processInfo.environment["HARNESS_EV"]!
    let root = ProcessInfo.processInfo.environment["HARNESS_ROOT"]!
    var log = ""

    func note(_ s: String) { log += s + "\n"; print("HARNESS: " + s) }

    func pump(_ seconds: TimeInterval) {
        RunLoop.main.run(until: Date().addingTimeInterval(seconds))
    }

    func render(_ model: FloaterModel, _ name: String) {
        let view = FloaterView(model: model, mover: WindowMover())
            .padding(10)
            .background(Color(red: 0.2, green: 0.24, blue: 0.3))
        let r = ImageRenderer(content: view)
        r.scale = 4
        guard let img = r.nsImage, let tiff = img.tiffRepresentation,
              let rep = NSBitmapImageRep(data: tiff),
              let png = rep.representation(using: .png, properties: [:]) else {
            note("render failed for \(name)"); return
        }
        try? png.write(to: URL(fileURLWithPath: "\(ev)/\(name).png"))
        note("rendered \(name).png  status=\"\(model.status)\" shotCount=\(model.shotCount) screenTrusted=\(model.screenTrusted)")
    }

    func makeHome(capture: String) -> String {
        let home = NSTemporaryDirectory() + "fm-harness-\(UUID().uuidString.prefix(8))"
        try! FileManager.default.createDirectory(atPath: home, withIntermediateDirectories: true)
        let cap = home + "/cap"
        try! capture.write(toFile: cap, atomically: true, encoding: .utf8)
        chmod(cap, 0o755)
        setenv("FM_DESK_SHOT_CAPTURE", cap, 1)
        setenv("FM_STATE_OVERRIDE", home + "/state", 1)
        return home
    }

    func inbox(_ home: String) -> [[String: Any]] {
        let dir = home + "/state/desk-voice/inbox"
        let files = ((try? FileManager.default.contentsOfDirectory(atPath: dir)) ?? []).filter { $0.hasSuffix(".json") }.sorted()
        return files.compactMap {
            (try? JSONSerialization.jsonObject(with: Data(contentsOf: URL(fileURLWithPath: dir + "/" + $0)))) as? [String: Any]
        }
    }

    let okCapture = """
    #!/usr/bin/env bash
    echo "$*" >> "$(dirname "$0")/cap.log"
    for a in "$@"; do out=$a; done
    printf 'PNG' > "$out"
    """

    func testStackedShotsSendAsOneMessage() {
        let home = makeHome(capture: okCapture)
        let model = FloaterModel(repoRoot: root, fmHome: home)
        model.keysTrusted = true
        model.screenTrusted = true
        render(model, "floater-idle")
        model.takeShot()
        pump(0.6)
        note("after shot 1: status=\"\(model.status)\" shotCount=\(model.shotCount) inbox=\(inbox(home).count)")
        model.takeShot()
        pump(0.6)
        model.takeShot()
        pump(0.6)
        note("after shot 3: status=\"\(model.status)\" shotCount=\(model.shotCount) inbox=\(inbox(home).count)")
        render(model, "floater-3-shots-stacked")
        XCTAssertEqual(model.shotCount, 3)
        XCTAssertEqual(model.status, "3 shots stacked")
        XCTAssertEqual(inbox(home).count, 0, "nothing sent while still stacking")
        pump(2.0)
        note("2s after last shot (inside 3s window): inbox=\(inbox(home).count)")
        XCTAssertEqual(inbox(home).count, 0)
        pump(2.0)
        let msgs = inbox(home)
        note("4s after last shot: inbox=\(msgs.count) status=\"\(model.status)\" shotCount=\(model.shotCount)")
        render(model, "floater-sent")
        XCTAssertEqual(msgs.count, 1)
        XCTAssertEqual((msgs.first?["images"] as? [String])?.count, 3)
        XCTAssertEqual(msgs.first?["source"] as? String, "desk-floater")
        note("message: \(msgs.first?["transcript"] as? String ?? "nil")")
        note("capture calls:\n" + ((try? String(contentsOfFile: home + "/cap.log", encoding: .utf8)) ?? ""))
        XCTAssertEqual(model.status, "Sent")
        XCTAssertEqual(model.shotCount, 0)
        // A later single shot goes on its own.
        model.takeShot()
        pump(3.8)
        let after = inbox(home)
        note("single later shot: inbox=\(after.count) last=\(after.last?["transcript"] as? String ?? "nil")")
        XCTAssertEqual(after.count, 2)
        XCTAssertEqual((after.last?["images"] as? [String])?.count, 1)
        try? log.write(toFile: "\(ev)/floater-model-harness-stack.txt", atomically: true, encoding: .utf8)
    }

    func testWithoutScreenRecordingNoShotIsTaken() {
        let home = makeHome(capture: okCapture)
        let model = FloaterModel(repoRoot: root, fmHome: home)
        model.keysTrusted = true
        model.screenTrusted = false
        render(model, "floater-screen-off-badge")
        var asked = 0
        model.onFixScreen = { asked += 1 }
        model.takeShot()
        pump(0.3)
        note("takeShot without permission: status=\"\(model.status)\" shotCount=\(model.shotCount) captureCalled=\(FileManager.default.fileExists(atPath: home + "/cap.log"))")
        XCTAssertEqual(model.status, "Screen off - click !")
        XCTAssertFalse(FileManager.default.fileExists(atPath: home + "/cap.log"))
        model.shotButton()
        note("camera click without permission asks macOS again: onFixScreen called \(asked)x")
        XCTAssertEqual(asked, 1)
        pump(3.5)
        XCTAssertEqual(inbox(home).count, 0)
        note("inbox after 3.5s: \(inbox(home).count)")
        try? log.write(toFile: "\(ev)/floater-model-harness-screen-off.txt", atomically: true, encoding: .utf8)
    }

    func testFailedCaptureSendsNothing() {
        let home = makeHome(capture: "#!/usr/bin/env bash\necho \"$*\" >> \"$(dirname \"$0\")/cap.log\"\nexit 1\n")
        let model = FloaterModel(repoRoot: root, fmHome: home)
        model.keysTrusted = true
        model.screenTrusted = true
        model.takeShot()
        note("failed capture, while capturing: status=\"\(model.status)\" shotCount=\(model.shotCount)")
        var waited = 0.0
        while model.shotCount > 0 && waited < 3 { pump(0.05); waited += 0.05 }
        note("failed capture: status=\"\(model.status)\" shotCount=\(model.shotCount)")
        XCTAssertEqual(model.status, "Shot failed")
        XCTAssertEqual(model.shotCount, 0)
        pump(3.5)
        note("after 4s: inbox=\(inbox(home).count) status=\"\(model.status)\"")
        XCTAssertEqual(inbox(home).count, 0)
        try? log.write(toFile: "\(ev)/floater-model-harness-failed-capture.txt", atomically: true, encoding: .utf8)
    }
}
